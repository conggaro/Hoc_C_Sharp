    static string ToUnicodeEscape(string input)
    {
        StringBuilder sb = new StringBuilder();
        foreach (char c in input)
        {
            if (c > 127) // only escape non-ASCII characters
            {
                sb.AppendFormat("\\{0:X4}", (int)c);
            }
            else
            {
                sb.Append(c);
            }
        }
        return sb.ToString();
    }

    static string FromUnicodeEscape(string input)
    {
        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < input.Length; i++)
        {
            if (input[i] == '\\' && i + 5 < input.Length && input[i + 1] == 'u' && IsHex(input, i + 2, 4))
            {
                string hex = input.Substring(i + 2, 4);
                int code = Convert.ToInt32(hex, 16);
                sb.Append((char)code);
                i += 5;
            }
            else if (input[i] == '\\' && i + 4 < input.Length && IsHex(input, i + 1, 4))
            {
                string hex = input.Substring(i + 1, 4);
                int code = Convert.ToInt32(hex, 16);
                sb.Append((char)code);
                i += 4;
            }
            else
            {
                sb.Append(input[i]);
            }
        }

        return sb.ToString();
    }

    static bool IsHex(string input, int startIndex, int length)
    {
        for (int i = startIndex; i < startIndex + length; i++)
        {
            char c = input[i];
            bool isHex =
                (c >= '0' && c <= '9') ||
                (c >= 'A' && c <= 'F') ||
                (c >= 'a' && c <= 'f');

            if (!isHex)
            {
                return false;
            }
        }

        return true;
    }
